cmake_minimum_required(VERSION 3.24)


# ============================================================================
# Project
# ============================================================================
#
project(OpenGL
	VERSION 0.1.0
	LANGUAGES CXX C
)


# ============================================================================
# Global Cmake settings
# ============================================================================
#
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF) # строгий стандарт ISO C++ для всех целей (отключить использование специфичных функций конкретного компилятора)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON) # генерация файла compile_commands.json (для подсвечивания синтаксиса, навигации, автодополнения и статического анализа VS Code)


# ============================================================================
# Dependencies 
# ============================================================================
#

include(FetchContent)

# ----------------------------------------------------------------------------
# GLFW
# ----------------------------------------------------------------------------
#
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS    OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS     OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL        OFF CACHE BOOL "" FORCE)

FetchContent_Declare(
	glfw
	GIT_REPOSITORY https://github.com/glfw/glfw.git
	GIT_TAG 3.5.1
)

FetchContent_MakeAvailable(glfw)


# ----------------------------------------------------------------------------
# GLM 
# ----------------------------------------------------------------------------
#
set(GLM_BUILD_TESTS OFF CACHE BOOL "" FORCE)

FetchContent_Declare(
	glm
	GIT_REPOSITORY https://github.com/g-truc/glm.git
	GIT_TAG 1.0.3
)

FetchContent_MakeAvailable(glm)


# ----------------------------------------------------------------------------
# fastgltf
# ----------------------------------------------------------------------------
#
set(FASTGLTF_ENABLE_TESTS    OFF CACHE BOOL "" FORCE)
set(FASTGLTF_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FASTGLTF_ENABLE_DOCS     OFF CACHE BOOL "" FORCE)
set(FASTGLTF_ENABLE_GLTF_RS  OFF CACHE BOOL "" FORCE)
set(FASTGLTF_ENABLE_ASSIMP   OFF CACHE BOOL "" FORCE)
set(FASTGLTF_ENABLE_INSTALL  OFF CACHE BOOL "" FORCE)

FetchContent_Declare(
	fastgltf
	GIT_REPOSITORY https://github.com/spnda/fastgltf.git
	GIT_TAG v0.9.0
)

FetchContent_MakeAvailable(fastgltf)


# ----------------------------------------------------------------------------
# Assimp
# ----------------------------------------------------------------------------
#
set(BUILD_SHARED_LIBS                     OFF CACHE BOOL "" FORCE)

set(ASSIMP_BUILD_TESTS                    OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ASSIMP_TOOLS             OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_SAMPLES                  OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_DOCS                     OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL                        OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL_PDB                    OFF CACHE BOOL "" FORCE)

set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)

# Включаем импорт только нужных 3D-форматов (.obj, .gltf/.glb, .fbx)
set(ASSIMP_BUILD_OBJ_IMPORTER  ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_GLTF_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_FBX_IMPORTER  ON CACHE BOOL "" FORCE)

FetchContent_Declare(
	assimp
	GIT_REPOSITORY https://github.com/assimp/assimp.git
	GIT_TAG v6.0.5
)

FetchContent_MakeAvailable(assimp)






# ============================================================================
# Local dependencies
# ============================================================================
#


# ----------------------------------------------------------------------------
# GLAD
# ----------------------------------------------------------------------------
#
add_library(glad STATIC
	external/glad/src/glad.c
)

target_include_directories(glad
	PUBLIC
		${CMAKE_CURRENT_SOURCE_DIR}/external/glad/include
)


# ----------------------------------------------------------------------------
# stb_image
# ----------------------------------------------------------------------------
#
add_library(stb_image STATIC
	external/stb/stb_image.cpp
)

target_include_directories(stb_image
	PUBLIC
		${CMAKE_CURRENT_SOURCE_DIR}/external/stb
)







# ============================================================================
# Engine
# ============================================================================
#
file(GLOB_RECURSE ENGINE_SOURCES
	CONFIGURE_DEPENDS
	"${CMAKE_CURRENT_SOURCE_DIR}/engine/*.cpp"
	"${CMAKE_CURRENT_SOURCE_DIR}/engine/*.c"
)

file(GLOB_RECURSE ENGINE_HEADERS
	CONFIGURE_DEPENDS
	"${CMAKE_CURRENT_SOURCE_DIR}/engine/*.hpp"
	"${CMAKE_CURRENT_SOURCE_DIR}/engine/*.h"
)

add_library(engine STATIC
	${ENGINE_SOURCES}
	${ENGINE_HEADERS}
)

target_include_directories(engine
	PUBLIC
		${CMAKE_CURRENT_SOURCE_DIR}/engine
)

target_link_libraries(engine
	PUBLIC
		glfw
		glad
		stb_image
		glm::glm
		fastgltf::fastgltf
		assimp
)



# ============================================================================
# Engine compiler settings
# ============================================================================
#
if(MSVC)

	target_compile_options(engine
		PRIVATE
			/MP
		    /W4
			/wd26495
			/external:anglebrackets
            /external:W0
	)

endif()







# ============================================================================
# Sandbox
# ============================================================================
#
file(GLOB_RECURSE SANDBOX_SOURCES
	CONFIGURE_DEPENDS
	"${CMAKE_CURRENT_SOURCE_DIR}/sandbox/*.cpp"
	"${CMAKE_CURRENT_SOURCE_DIR}/sandbox/*.hpp"
	"${CMAKE_CURRENT_SOURCE_DIR}/sandbox/*.h"
)

add_executable(sandbox 
	${SANDBOX_SOURCES}
)

target_link_libraries(sandbox
	PRIVATE
		engine
)

# ============================================================================
# Sandbox compiler settings
# ============================================================================
#
if(MSVC)

	target_compile_options(sandbox
		PRIVATE
			/W4
            /MP
	)
    
	set_target_properties(sandbox PROPERTIES # меняет Working Directory на дерикторию где лежит исполняемый файл .exe
		VS_DEBUGGER_WORKING_DIRECTORY "$<TARGET_FILE_DIR:sandbox>"
	)

endif()



# ============================================================================
# Resources
# ============================================================================
#

set(RESOURCE_DIRECTORY
    "${CMAKE_CURRENT_SOURCE_DIR}/Resources"
)

add_custom_target(CopyResources ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${RESOURCE_DIRECTORY}"
        "$<TARGET_FILE_DIR:sandbox>/Resources"

    COMMENT "Copying resources..."
)

add_dependencies(sandbox CopyResources)



# ============================================================================
# Information
# ============================================================================
#
message(STATUS "")
message(STATUS "========================================")
message(STATUS " OpenGL Engine")
message(STATUS "========================================")
message(STATUS " C++ standard : C++${CMAKE_CXX_STANDARD}")
message(STATUS " Compiler     : ${CMAKE_CXX_COMPILER_ID}")
message(STATUS " Build type   : ${CMAKE_BUILD_TYPE}")
message(STATUS "========================================")
message(STATUS "")
