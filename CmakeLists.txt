cmake_minimum_required(VERSION 3.16)


# --- Project Settings --- #

project(OpenGL CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)






# --- External Libraries --- #

# GLFW
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(external/glfw)

# GLAD
add_library(glad STATIC external/glad/src/glad.c)
target_include_directories(glad PUBLIC external/glad/include)





# --- Executable --- #
	
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS
	"src/*.cpp"
	"src/*.c"
)

file(GLOB_RECURSE HEADERS CONFIGURE_DEPENDS
	"src/*.h"
	"src/*.hpp"
)

#source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR}/src PREFIX "src" FILES ${SOURCES} ${HEADERS})

add_executable(${PROJECT_NAME} ${SOURCES} ${HEADERS})

add_compile_definitions(
	GLM_FORCE_INLINE                   # Принудительный инлайнинг всех математических функций
    GLM_FORCE_INTRINSICS               # Включение SIMD (SSE2/AVX/NEON)
    GLM_FORCE_DEFAULT_ALIGNED_GENTYPES # Выравнивание данных в памяти по 16 байт (для SIMD)
)

target_compile_definitions(${PROJECT_NAME} PRIVATE
	GLM_ENABLE_EXPERIMANTAL
)

if(MSVC)
	target_compile_options(${PROJECT_NAME} PRIVATE
		$<$<OR:$<CONFIG:Release>,$<CONFIG:RelWithDebInfo>>:/02;/0i;/0t;/fp-fast>
		$<$<CONFIG:Debug>:/0d;/RTC1;/Zi>
	)
endif()


target_include_directories(${PROJECT_NAME} PRIVATE
	external/glad/include
	external/glm
	${CMAKE_CURRENT_SOURCE_DIR}/src
)


# Линкуем  glfw и glad (без OpenGL::GL)
target_link_libraries(OpenGL PRIVATE glfw glad)

# Copy /Resources dir to dir with .exe file
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
	COMMAND ${CMAKE_COMMAND} -E copy_directory
	"${CMAKE_CURRENT_SOURCE_DIR}/Resources"
	"$<TARGET_FILE_DIR:${PROJECT_NAME}>/Resources"
)
